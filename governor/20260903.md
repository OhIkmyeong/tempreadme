export class SubEasyGbusView {
    /**
    * 레이어 팝업을 만들어 엽니다
    * @param {Object}
    * - isMine {Boolean} 내가 작석한 글인지
    * - addressStart {String} 기점 주소
    * - detailStart {String} 기점 상세주소
    * - addressEnd {String} 종점 주소
    * - detailEnd {String} 종점 상세주소
    * - timeStart {String} HH:MM
    * - timeEnd {String} HH:MM
    * - reason {String} 제안이유
    * - date {String}YYYY.MM.DD
    * - num {String|Number}
    */
    static open_layer_popup = (INFO = {}) => {
        const { isMine, addressStart, detailStart, addressEnd, detailEnd, timeStart, timeEnd, reason, date, num } = INFO

        /* class */
        const clss = `layerpop`;

        /* DOM */
        const $wrap = document.createElement("DIV");
        $wrap.classList.add(`${clss}-wrap`);

        /* 모달(레이어팝업) */
        const $pop = document.createElement("SECTION");
        $pop.classList.add(`${clss}`);
        $wrap.appendChild($pop);

        /* 닫기 버튼 */
        const $close = document.createElement("BUTTON");
        $close.classList.add(`${clss}-close`);
        $close.type = "button";
        $close.setAttribute("aria-label", "제안 상세보기 닫기");
        // $close.textContent = "닫기";
        $pop.appendChild($close);

        /* 닫으면 삭제 */
        $close.addEventListener("click", () => {
            /* 포커스 트랩 제거 */
            $wrap.removeEventListener("keydown", ModalCtrl.focus_trap_modal);
            /* 윈도우 스크롤 관련 */
            OverflowCtrl.allowOverFlow();
            /* 삭제 */
            $wrap.remove();
        }, { once: true });

        /* 제목 */
        const $title = document.createElement("H1");
        $title.classList.add(`${clss}-title`);
        $title.textContent = "제안 상세보기";
        $pop.appendChild($title);

        /* 본문 */
        $pop.appendChild(SubEasyGbusView.content(INFO, clss));

        /* last */
        document.body.appendChild($wrap);

        /* 윈도우 스크롤 관련 */
        OverflowCtrl.stopOverFlow();

        /* 포커스 트랩 추가 */
        $close.focus();
        $wrap.addEventListener("keydown", ModalCtrl.focus_trap_modal);
    }//open_layer_popup

    /**
     * 
     * @param {*} INFO 
     */
    static content = (INFO, clss) => {
        const { isMine, addressStart, detailStart, addressEnd, detailEnd, timeStart, timeEnd, reason, date, num } = INFO

        /* DOM */
        const $frag = document.createDocumentFragment();

        /* 영역 만들기 */
        const make_area = (title, key) => {
            const $area = document.createElement("ARTICLE");
            $area.classList.add(`${clss}-area`);
            $area.classList.add(`${clss}-area__${key}`);
            if (title) {
                DomCommonView.add_p_for_wrap({
                    $target: $area,
                    content: [title],
                    clss: `${clss}-area-title`,
                    tagAlt: "H3"
                });
            }
            return $area;
        }//make_area

        const make_borderbox = (key) => {
            const $borderbox = document.createElement("DIV");
            $borderbox.classList.add(`${clss}-bbox`);
            $borderbox.classList.add(`${clss}-bbox__${key}`);
            return $borderbox;
        }

        /* 거점/종점 */
        {
            const key = "addr";
            const $area = make_area("거점/종점", key);
            const $box = make_borderbox("addr");
            $area.appendChild($box);
            $frag.appendChild($area);
            [
                {
                    lbl: "기점",
                    addr: addressStart,
                    detail: detailStart
                },
                {
                    lbl: "종점",
                    addr: addressEnd,
                    detail: detailEnd
                }
            ].forEach(item => {
                const $item = document.createElement("DIV");
                $item.classList.add(`${clss}-${key}-row`);
                $box.appendChild($item);

                ["lbl", "addr", "detail"].forEach(okey => {
                    DomCommonView.add_p_for_wrap({
                        $target: $item,
                        content: [item[okey]],
                        clss: `${clss}-addr-row-${okey}`,
                        tagAlt: "SPAN"
                    });
                });
            });
        }//기점종점

        /* 희망시간 */
        {
            const key = "htime";
            const $area = make_area("희망시간", key);
            $frag.appendChild($area);

            [
                {
                    title: "출근(도착)",
                    time: timeStart
                }, {
                    title: "퇴근(출발)",
                    time: timeEnd
                }
            ].forEach(item => {
                const $box = make_borderbox(key);
                $area.appendChild($box);
                ["title", "time"].forEach(okey => {
                    DomCommonView.add_p_for_wrap({
                        $target: $box,
                        content: [item[okey]],
                        clss: `${clss}-htime-row-${okey}`,
                        tagAlt: "SPAN"
                    });
                });
            });
        }//희망시간

        {
            const key = "rsn"
            const $area = make_area("제안이유", key);
            $frag.appendChild($area);

            const $box = make_borderbox(key);
            $box.textContent = reason;
            $area.appendChild($box);
        }

        /* 기타 정보 */
        {

            const $area = make_area("", key);
            $frag.appendChild($area);

            DomCommonView.add_p_for_wrap({

                    `신청일 ${date}`,
                    `접수번호 ${num}`
                ],
                clss: `${clss}-${key}-elem`
            });
