# EventBnrCtrl.js (수정)
경로 : js/controller/components/EventBnrCtrl.js
```js
/**
 * 메인페이지 > 이벤트 배너 슬라이더 컨트롤
 */
export class EventBnrCtrl {
    constructor() {
        /* DOM */
        this.$wrap;
        this.$slider;
        this.$$item;
        this.$curr;
        this.$total;
        this.$prev;
        this.$play;
        this.$next;
        /* 상수 */
        this.LEN;
        this.CURR = 1;
        /* progress bar */
        this.PER_UNIT;
        /* 리사이즈 타이머 */
        this.resizeTimer;
        /* touch,click */
        this.startX;
        this.startY;
        this.posX;
        this.nowPosX;
        this.isHorizontalMove = null;
        /* 자동재생 타이머 */
        this.autoPlayTimer = null;
        this.isPaused = false;
        this.autoplayDelay = 3000;
    }//constructor

    /**
     * 
     */
    init() {
        /* early return dev */
        if (this.off_when_dev()) { return; };

        /* DOM 설정 */
        this.set_dom();
        if (!this.$slider) {
            requestAnimationFrame(() => {
                this.init();
            });
            return;
        }

        /* 만일 아이템이 1개라면 early return */
        if (this.LEN <= 1) {
            this.$pager.style.display = "none";
            return;
        } else {
            /* bar를 위한 per unit 설정 */
            this.PER_UNIT = 1 / this.LEN;
        }

        /* 첫번째, 마지막 슬라이드 붙여넣기 해야함 */
        this.copy_first_last_slides();

        /* 이전, 다음 버튼 클릭시 */
        this.$prev.addEventListener("click", () => {
            this.on_click_prev_next(-1);
        });
        this.$next.addEventListener("click", () => {
            this.on_click_prev_next(1);
        });

        /* 리셋 */
        this.reset();

        /* 클릭 + 터치 이벤트 시작 */
        this.toggle_start_event(true);

        /* 리사이즈 타이머 */
        this.lastWindowWidth = window.innerWidth;
        window.addEventListener("resize", () => {
            /* 주소창 올라감/내려감으로 인한 height 변경은 무시하고, 실제 width가 변경되었을 때만 처리 */
            const currentWidth = window.innerWidth;
            if (currentWidth === this.lastWindowWidth) {
                return;
            } else {
                this.lastWindowWidth = currentWidth;
            }

            /* 타이머 취소 */
            clearTimeout(this.resizeTimer);

            this.resizeTimer = setTimeout(() => {
                this.reset();
                this.stop_autoplay();
            }, 300);
        });

        /* 자동재생 타이머 */
        this.$play.addEventListener("click", this.toggle_autoplay);
        this.start_autoplay();
    }//init

    /**
     * 개발모드에선 꺼두기
     */
    off_when_dev() {
        if (window.location.pathname.includes("_dev.html")) {
            console.log("_dev.html 에선 작동하지 않습니다.");
            return true;
        }
        return false;
    }//off_when_dev

    /**
     * 
     */
    reset = () => {
        this.CURR = 1;
        this.toggle_transition(false);
        this.display_bar();
        this.go_to();
    }//reset


    /**
     * 
     */
    set_dom = () => {
        this.$wrap = document.querySelector(".mnevtbnr-wrap");
        this.$slider = document.querySelector(".mnevtbnr");
        this.$$item = document.querySelectorAll(".mnevtbnr-item");

        this.$bar = document.querySelector(".mnevtbnr-prgs-bar");

        this.LEN = this.$$item.length;
        this.$curr = document.querySelector(".mnevtbnrpager-idx.curr");
        this.$curr.textContent = String(this.CURR).padStart(2, "0");

        this.$pager = document.querySelector(".mnevtbnrpager");
        this.$total = document.querySelector(".mnevtbnrpager-idx.total");
        this.$total.textContent = String(this.LEN).padStart(2, "0");

        this.$prev = document.querySelector(".mnevtbnrpager-btn-prev");
        this.$next = document.querySelector(".mnevtbnrpager-btn-next");
        this.$play = document.querySelector(".mnevtbnrpager-btn-play");
    }//set_dom

    /**
     * 첫번째,마지막 슬라이드 앞뒤로 붙여주기
     */
    copy_first_last_slides() {
        const $first = this.$$item[0].cloneNode(true);
        $first.dataset.idx = this.LEN;
        this.$slider.appendChild($first);

        const $last = this.$$item[this.LEN - 1].cloneNode(true);
        $last.dataset.idx = -1;
        this.$slider.prepend($last);
    }//copy_first_last_slides

    /**
     * 슬라이더의 트랜지션 정하기
     * @param {Boolean} bool true시 트랜지션 가능 
     */
    toggle_transition(bool) {
        const clss = "notransition";
        if (bool) {
            this.$slider.classList.remove(clss);
        } else {
            this.$slider.classList.add(clss);
        }
    }//toggle_transition

    /**
     * 슬라이더의 링크 클릭 가능 여부 정하기
     * @param {Boolean} bool true시 링크 클릭 가능 
     */
    toggle_click(bool) {
        const clss = "noclick";
        if (bool) {
            this.$slider.classList.remove(clss);
        } else {
            this.$slider.classList.add(clss);
        }
    }//toggle_transition

    /**
     * 
     */
    go_to() {
        const wid = this.$wrap.getBoundingClientRect().width;
        this.$slider.style.transform = `translateX(${-1 * wid * this.CURR}px)`
        /* 바 변경 */
        this.display_bar();
    }//go_to

    /**
     * 바 변경 및 curr text 변경
     */
    display_bar() {
        let barIdx = this.CURR;
        if (this.CURR <= 0) {
            barIdx = this.LEN;
        } else if (this.CURR > this.LEN) {
            barIdx = 1;
        }
        const per = Math.min(1, this.PER_UNIT * barIdx);
        this.$bar.style.setProperty("--_per", per);

        this.$curr.textContent = String(barIdx).padStart(2, "0");
    }//display_bar

    /**
     * 
     * @param {*} diff 
     */
    on_click_prev_next = (diff) => {
        /* 자동 재생 인터벌 끄기 */
        this.stop_autoplay();

        /* 버튼 판별 */
        const $btn = diff > 0 ? this.$next : this.$prev;
        $btn.disabled = true;

        this.CURR += diff;
        this.toggle_transition(true);
        this.go_to();

        this.$slider.addEventListener("transitionend", () => {
            const over = this.CURR > this.LEN;
            const less = this.CURR <= 0;

            if (over || less) {
                this.toggle_transition(false);
                this.cacul_curr();
                this.go_to();
                setTimeout(() => {
                    this.toggle_transition(true);
                }, 100);
            }

            $btn.disabled = false;
        }, { once: true });
    }//on_click_prev_next

    /**
     * 클릭 + 터치 이벤트 시작할지 말지
     * @param {Boolean} add 
     */
    toggle_start_event = (add) => {
        const method = add ? 'addEventListener' : 'removeEventListener';
        this.$slider[method]("mousedown", this.on_start_down, { once: true });
        this.$slider[method]("touchstart", this.on_start_down, { once: true });
    }//toggle_start_event

    /**
     * 좌표값 가져오기
     * @param {Event} e 
     */
    get_coords = (e) => {
        const result = { x: 0, y: 0 };
        if (e.type.startsWith('touch')) {
            result.x = e.touches[0].clientX;
            result.y = e.touches[0].clientY;
        } else {
            result.x = e.clientX;
            result.y = e.clientY;
        }
        return result;
    }//get_coords

    /**
     * 클릭 / 터치 시작
     * @param {*} e 
     */
    on_start_down = (e) => {
        /* 인앱브라우저 고스트 클릭 문제 */
        if (e.type.startsWith('touch')) {
            this._usedTouch = true;
        } else {
            this._usedTouch = false;
        }
        if (e.type === 'mousedown' && this._usedTouch) return; // ghost mousedown 무시

        /*  */
        const { x, y } = this.get_coords(e);
        const { left } = this.$wrap.getBoundingClientRect();

        // console.log(left);

        /* 자동재생 인터벌 끄기 */
        this.stop_autoplay();

        /* 시작 */
        this.startX = x;
        this.startY = y;

        /* 판별 변수 초기화 */
        this.isHorizontalMove = null;

        /* 스크롤 left */
        const scrollLeft = window.scrollX || document.documentElement.scrollLeft;

        /* 위치 */
        this.posX = x;
        // console.log("posX",this.posX);

        /* 트랜지션 끄기 */
        this.toggle_transition(false);

        /* move,end 이벤트 바인딩 */
        if (e.type.startsWith('touch')) {
            window.addEventListener('touchmove', this.on_move, { passive: false });
            window.addEventListener('touchend', this.cancel, { once: true });
            window.addEventListener('touchcancel', this.cancel, { once: true });
        } else {
            window.addEventListener('mousemove', this.on_move);
            window.addEventListener('mouseup', this.cancel, { once: true });
            window.addEventListener('mouseleave', this.cancel, { once: true });
        }
    }//on_start_down

    /**
     * 움직일
     * @param {Event} e 
     */
    on_move = (e) => {
        const { x, y } = this.get_coords(e);
        const { width } = this.$wrap.getBoundingClientRect();

        /* 터치일때 스크롤 방지 */
        const diffX = Math.abs(x - this.startX);
        const diffY = Math.abs(y - this.startY);

        // 1. 처음 움직임이 감지되었을 때 방향을 한 번만 판별합니다.
        if (this.isHorizontalMove === null) {
            // 최소 5~10px 정도 움직였을 때 판별 시작 (미세한 떨림 방지)
            if (diffX > 5 || diffY > 5) {
                this.isHorizontalMove = diffX > diffY;
            }
        }

        // 2. 좌우 이동이라고 판단되면 브라우저 스크롤을 막습니다.
        if (this.isHorizontalMove === true) {
            if (e.cancelable) e.preventDefault();
        } else if (this.isHorizontalMove === false) {
            // 상하 이동이라고 판단되면 슬라이드 로직을 타지 않고 리턴 (브라우저 스크롤 허용)
            this.nowPosX = 0;
            return;
        }

        /* 계산 시작 */
        this.nowPosX = x - this.posX;
        let moveX = (this.CURR * -1 * width) + this.nowPosX;

        const finalX = () => {
            if (moveX >= 0) { return 0; }
            return moveX;
        }

        /* 클릭 방지 */
        if (Math.abs(this.nowPosX) > window.innerWidth / 10) {
            this.toggle_click(false);
        }

        /* 움직이기 */
        this.$slider.style.transform = `translateX(${finalX()}px)`
    }//on_move

    /**
     * 
     */
    cancel = () => {
        /* 클릭 방지 해제 */
        this.toggle_click(true);

        /* 트랜지션 가능 */
        this.toggle_transition(true);
        // console.log(this.CURR);

        /* 스냅핑 */
        if (Math.abs(this.nowPosX) >= this.$wrap.getBoundingClientRect().width / 4) {
            // console.log("스냅핑 변화", this.nowPosX);
            const diff = this.nowPosX < 0 ? 1 : -1;
            this.CURR = this.CURR + diff;
            this.go_to();

            this.$slider.addEventListener("transitionend", () => {
                this.toggle_transition(false);
                this.cacul_curr();
                this.go_to();
            }, { once: true });
        } else {
            // console.log("스냅핑 유지");
            this.go_to();
        }
        // console.log(this.CURR);
        this.nowPosX = 0;

        /* 이벤트 제거 */
        window.removeEventListener('mousemove', this.on_move);
        window.removeEventListener('touchmove', this.on_move);

        /* 다운/터치 스타투 이벤트 추가 */
        this.toggle_start_event(true);
    }//cancel

    /**
     * curr index 정상화
     */
    cacul_curr = () => {
        if (this.CURR > this.LEN) {
            this.CURR = 1;
        } else if (this.CURR <= 0) {
            this.CURR = this.LEN;
        }
    }//cacul_curr

    /**
     * 자동 재생 시작
     */
    start_autoplay = () => {
        console.log("시작 명령을 받음");
        /* 중복 실행 방지 */
        this.stop_autoplay();

        /* 버튼 토글 */
        this.$play.classList.remove("paused");

        /* 시작 */
        this.autoPlayTimer = setInterval(() => {
            if (this.isPaused) {
                this.stop_autoplay();
                return;
            }
            // console.log("setInterval");

            /* 트랜지션 가능하게 */
            this.toggle_transition(true);

            /* 계산 및 이동 */
            let nextIdx = this.CURR + 1;
            if (nextIdx > this.LEN) {
                nextIdx = 1;
            }
            this.CURR = nextIdx;
            // console.log(this.CURR);
            this.go_to();
        }, this.autoplayDelay);
    }//start_autoplay

    /**
     * 자동 재생 중지
     */
    stop_autoplay = () => {
        if (this.autoPlayTimer) {
            console.log("정지");
            clearInterval(this.autoPlayTimer);
            this.isPaused = true;
            this.autoPlayTimer = null;
            this.$play.classList.add("paused");
        }
    }//stop_autoplay

    /**
     * 자동재생 flag 토
     */
    toggle_autoplay = () => {
        const prev = this.isPaused;
        this.isPaused = !prev;

        if (this.isPaused) {
            this.stop_autoplay();
        } else {
            this.start_autoplay();
        }
    }//toggle_btn_autoplay
}//class:EventBnrCtrl
```
