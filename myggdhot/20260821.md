/**
 * 드래그 가능하게 만듦
 */
export class Draggable {
    constructor() {
        this.$drag = null;
        this.posX = null;
        this.posY = null;
    }
    /**
     * 
     * @param {*} $drag 
     * @returns 
     */
    set_dom($drag) {
        this.$drag = $drag;
        return this;
    }//set_dom

    /**
     * 
     */
    init() {
        this.toggle_start_events(true);
    }//init

    /**
     * 
     */
    toggle_start_events = (add) => {
        const method = add ? 'addEventListener' : 'removeEventListener';
        this.$drag[method]("mousedown", this.on_start, { once: true });
        this.$drag[method]("touchstart", this.on_start, { once: true, passive: false });
    }//add_mouse_down

    /**
     * Mouse/Touch 이벤트에서 좌표를 추출하는 공통 함수
     */
    get_coords(e) {
        const result = { x: 0, y: 0 };
        if (e.type.startsWith('touch')) {
            result.x = e.touches[0].pageX;
            result.y = e.touches[0].pageY;
        } else {
            result.x = e.pageX;
            result.y = e.pageY;
        }
        return result;
    }

    /**
     * 
     * @param {*} e 
     */
    on_start = (e) => {
        const excludes = ["A", "BUTTON", "LABEL", "INPUT"];
        if (excludes.includes(e.target.tagName)) {
            this.toggle_start_events(true);
            return;
        }
        // 터치 이벤트에서만 preventDefault를 실행하여 드래그에 집중하게 합니다.
        if (e.type === 'touchstart') {
            // e.cancelable 확인 후 실행 (안전장치)
            if (e.cancelable) e.preventDefault();
        }

        // 터치 시 스크롤 등 기본 동작 방지
        if (e.type === 'touchstart') { e.preventDefault(); }

        document.body.appendChild(this.$drag);

        const { x, y } = this.get_coords(e);
        const { left, top } = this.$drag.getBoundingClientRect();

        const scrollLeft = window.scrollX || document.documentElement.scrollLeft;
        const scrollTop = window.scrollY || document.documentElement.scrollTop;

        this.posX = x - (left + scrollLeft);
        this.posY = y - (top + scrollTop);

        // Move & End 이벤트 바인딩
        window.addEventListener('mousemove', this.on_move);
        window.addEventListener('touchmove', this.on_move, { passive: false });

        window.addEventListener('mouseup', this.cancel, { once: true });
        window.addEventListener('touchend', this.cancel, { once: true });
        window.addEventListener('mouseleave', this.cancel, { once: true });
    }//on_start

    /**
     * 
     * @param {*} e 
     */
    on_move = (e) => {
        const { x, y } = this.get_coords(e);
        const { width, height } = this.$drag.getBoundingClientRect();

        const nowPosX = x - this.posX;
        const nowPosY = y - this.posY;

        const clamp = (val, min, max) => Math.max(min, Math.min(val, max));

        const maxX = document.documentElement.scrollWidth - width;
        const maxY = document.documentElement.scrollHeight - height;

        const finalX = clamp(nowPosX, 0, maxX);
        const finalY = clamp(nowPosY, 0, maxY);

        this.$drag.style.transform = `translate(${finalX}px, ${finalY}px)`;
    }//on_mouse_move

    /**
     * 
     * @param {*} e 
     */
    cancel = (e) => {
        window.removeEventListener('mousemove', this.on_move);
        window.removeEventListener('touchmove', this.on_move);
        this.toggle_start_events(true);
    }//cancel
}//Draggable

            .init();

     * 
     * @param {*} $popup 
     */

        /* 팝업 보이게 */
        $popup.classList.remove("off");

        /* 닫기 버튼 클릭 이벤트 추가 */
        const $close = $popup.querySelector(".close");
        $close?.addEventListener("click", () => {
            this.on_close($popup);
        }, { once: true });
    }//add_close_popup


    /**
     * 
     * @param {*} $popup 
     */
    on_close = ($popup) => {
        /* 24h 체크 */
        const $check = $popup.querySelector(".check24");
        if ($check.checked) {
            this.set_cookie({
                popupid: $check.value,
                value: "done",
                expiredays: 1
            });
        }

        /* 닫기 */
        const $parent = $popup.parentElement;
        $parent.removeChild($popup);
    }//on_close

    /**
     * 쿠키 설정 관련 : 쿠키 설정
     * @param {*} param0 
     */
    set_cookie = ({ popupid, value, expiredays }) => {
        const today = new Date();
        today.setDate(today.getDate() + expiredays);
        document.cookie = popupid + "=" + encodeURIComponent(value) + "; path=/; expires=" + today.toGMTString() + ";"
    }//set_cookie

    /**
     * 쿠키 설정 관련 : 쿠키 가져오기
     */
    get_cookie = (popupid) => {
        const cookieData = document.cookie;

            return false;
        }

        if (!$popup) {
            throw new Error("can't find popup");
        }
        // console.log(`${popupid} 있음`);
        this.on_close($popup);
        return true;
    }//get_cookie

     * 쿠키 설정 관련 : 쿠키 삭제

export class DateParser {
    /**
     * 
     * @param {String} dateStr "YYYY-MM-DD"
     * @returns {String} "YYYY.MM.DD" 
     */
    static dash_to_dot(dateStr) {
        const date = new Date(dateStr);

        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');

        return `${year}.${month}.${day}`;
    }//dash_to_dot

    /**
     * Date 객체로부터
     * @param {Date} date 
     * @returns {Object}
     */
    static date_to_string(date) {
        if (!date) { date = new Date(); }
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return {
            year,
            month,
            day,
            dash: `${year}-${month}-${day}`,
            key: `${year}${month}${day}`
        }
    }//date_to_information

    /**
     * yyyy-mm-ddT00:00:00 형식의 문자열을 받아 Date 객체로 반환합니다.
     * @param {String} str 
     * @memo 아이폰 7미만 대응을 위함
     */
    static string_to_date(str) {
        if (!str) { throw new Error("no String for date parser"); }
        const { year, month, date, hour, minute, second } = DateParser.string_to_object(str);
        return new Date(year, month - 1, date, hour, minute, second);
    }//string_to_date

    /**
     * 
     * @param {*} str 
     * @returns {Object} values are integar
     * @memo month is strictly transformed to Integar. if you want to use as 'Date' Keep in mind that you have to subtract 1 from month value
     */
    static string_to_object(str) {
        if (!str) { throw new Error("no string value for Date String Parser") }
        if (str.includes("T")) {
            const [yyyymmdd, hhmmss] = str.split("T");
            const [yyyy, mm, dd] = yyyymmdd.split("-").map(s => parseInt(s));
            const [h, m, s] = hhmmss.split(":").map(s => parseInt(s));
            return {
                year: yyyy,
                month: mm,
                date: dd,
                hour: h,
                minute: m,
                second: s
            };
        } else {
            const [yyyy, mm, dd] = str.split("-").map(s => parseInt(s));
            return {
                year: yyyy,
                month: mm,
                date: dd,
                hour: 0,
                minute: 0,
                second: 0
            };
        }
    }//string_to_object

    /**
     * date로 부터 getDay 해온 정수값을 
     * 한글 요일 / 영어 요일로 반환
     * @param {} date Date 객체 (new Date)
     */
    static get_day_kr_eng(date) {
        if (!date) { throw new Error("Date value is needed"); }
        const day = date.getDay();
        let kr = null;
        let eng = null;
        let engShort = null;
        switch (day) {
            case 0: {
                kr = "일";
                eng = "sunday";
                engShort = "sun";
            } break;
            case 1: {
                kr = "월";
                eng = "monday";
                engShort = "mon";
            } break;
            case 2: {
                kr = "화";
                eng = "tuesday";
                engShort = "tue";
            } break;
            case 3: {
                kr = "수";
                eng = "wednesday";
                engShort = "wed";
            } break;
            case 4: {
                kr = "목";
                eng = "thursday";
                engShort = "thu";
            } break;
            case 5: {
                kr = "금";
                eng = "friday";
                engShort = "fri";
            } break;
            case 6: {
                kr = "토";
                eng = "saturday";
                engShort = "sat";
            } break;
        }
        return { kr, eng, engShort };
    }//get_day_kr_eng

    /**
     * 서버타임 가져오기
     * @returns 
     */
    static async get_server_time() {
        try {
            const response = await fetch(window.location.href, { method: 'HEAD' });
            const serverDateString = response.headers.get('Date'); // 서버의 GMT 시간
            const serverTime = new Date(serverDateString);
            // console.log("서버 시간:", serverTime);
            return serverTime;
        } catch (e) {
            return new Date();
        }
    }//get_server_time

}//DateParser

import { DateParser } from "./../../model/DateParser.js";
import { DomCommonView } from "./../common/DomCommonView.js";

export class LayerPopupView {
    constructor() {
        this.list = [
            {
                id: "lpop001",
                dateStart: "2026-08-20T09:00:00",
                dateEnd: "2026-09-10T09:00:00",
                imgs: [
                    {
                        url: "/cont/resources/myggd_hot/event/layerpopup_01.png",

                        isDeco: false,

                ],

                    {
                        title: "링크텍스트",
                        label: "이벤트 바로가기",
                        url: "https://www.gg.go.kr/",
                        start: "2026-08-25T09:00:00",
                        end: "2026-09-10T09:00:00",
                    }
                ]
            }
        ];
    }//constructor

    static get clss() {
        return "lpop";
    }

    /**
     * 
     */
    init = async () => {
        const now = await DateParser.get_server_time();
        const nowTime = now.getTime();

        for (const item of this.list) {
            const { dateStart, dateEnd } = item;
            const timeStart = DateParser.string_to_date(dateStart).getTime();
            const timeEnd = DateParser.string_to_date(dateEnd || "2999-12-31").getTime();

            /* early return:날짜 */
            if (nowTime < timeStart || nowTime > timeEnd) {
                continue;
            }

            /* DOM 추가 */
            document.body.appendChild(this.make({ item, nowTime }));
        }//for
    }//init

    /**
     * 
     * @param {*} item 
     */
    make({ item, nowTime }) {
        const { id, dateStart, dateEnd, imgs, alt, links } = item;
        const clss = LayerPopupView.clss;

        /* DOM */
        const $wrap = document.createElement("SECTION");
        $wrap.classList.add(`${clss}`);
        $wrap.dataset.popupid = id;
        $wrap.classList.add(`off`);

        /* 이미지 영역 */
        const $visual = document.createElement("ARTICLE");
        $visual.classList.add(`${clss}-visual`);
        $visual.setAttribute("draggable", "false");
        $wrap.appendChild($visual);

        /* 이미지 채워넣기 */
        imgs.forEach((img, idx) => {
            const $img = DomCommonView.make_img(img);
            $img.classList.add(`${clss}-visual-img`);
            $img.classList.add(`${clss}-visual-img-${id}`);
            $img.classList.add(`${clss}-visual-img-${id}-${idx + 1}`);
            $img.setAttribute("draggable", "false");
            $visual.appendChild($img);
        });

        /* 링크 있으면 넣고 */
        if (links && links.length) {
            links.forEach((link, idx) => {
                const { url, title, label, start, end } = link;
                const start_time = DateParser.string_to_date(start).getTime();
                const end_time = DateParser.string_to_date(end).getTime();

                const linkStart = start_time <= nowTime;
                const linkEnd = end_time <= nowTime;
                const linkShow = linkStart && !linkEnd;

                const $link = document.createElement("A");
                $link.classList.add(`${clss}-link`);
                $link.classList.add(`${clss}-link-${id}`);
                $link.classList.add(`${clss}-link-${id}-${idx + 1}`);
                $link.target = "_blank";
                $link.href = url || "#";
                $link.innerHTML = title;
                $link.setAttribute("aria-label", `(새 창 열기)${label}`);
                if (!linkShow) {
                    $link.classList.add(`off`);
                }
                $visual.appendChild($link);
            });
        }

        /* 24시간 쿠키영역 */

        return $wrap;

    cookieArea({ id }) {
        const clss = LayerPopupView.clss;

        const $wrap = document.createElement("DIV");
        $wrap.classList.add(`${clss}-cookiewrap`);
        $wrap.classList.add(`${clss}-cookiewrap-${id}`);
        $wrap.setAttribute("draggable", "false");

        /* label */
        const $lbl = document.createElement("LABEL");
        $lbl.classList.add(`${clss}-lbl`);
        $wrap.appendChild($lbl);

        /* check */
        const $check = document.createElement("INPUT");
        $check.type = "checkbox";
        $check.classList.add(`${clss}-chk`);
        $check.classList.add(`check24`);
        $check.value = id;
        $lbl.appendChild($check);

        /* check:ic */
        const $ic = document.createElement("SPAN");
        $ic.classList.add(`${clss}-chkic`);
        $lbl.appendChild($ic);

        /* check:text */
        const $txt = document.createElement("SPAN");
        $txt.classList.add(`${clss}-chktxt`);
        $txt.textContent = "오늘 하루 보지 않기";
        $lbl.appendChild($txt);

        /* button */
        const $close = document.createElement("BUTTON");
        $close.type = "button";
        $close.dataset.closeId = id;
        $close.classList.add(`${clss}-close`);
        $close.classList.add(`close`);
        $close.textContent = "닫기";
        $wrap.appendChild($close);

        return $wrap;
    }//cookieArea
}//class:LayerPopupView
