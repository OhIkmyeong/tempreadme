/**
 * 하단 띠배너 컨트롤러
 */
export class StripBnrCtrl {
    constructor() {
        this.$$close = null;
        this.$ftr = null;
        this.keyLS = "gnhstripbnr";
    }
    /**
     * 하단 띠배너를 x 클릭하면 닫게하고 일주일 쿠키 적용
     * @memo 쿠키 방식으로 변경 2026-08-25
     * @memo 사용자단 쿠키삭제
     * Win : Ctrl + Shift + Delete
     * Mac: Cmd + Shift + Delete
     */
    init = () => {
        /* DOM setting */
        this.set_dom();

        /* 쿠키 가져오기 */
        this.load_cookies();
        // this.delete_cookie("stripbnr_about");

        /* 닫기 버튼 클릭시 */
        this.$$close.forEach(this.add_on_click_close);

        /* 스크롤시 */
        window.addEventListener("scroll", this.on_scroll);
    }//init

    /**
    * 로컬 스토리지 방식으로 열기
    */
    init_ls = () => {
        /* DOM setting */
        this.set_dom();

        /* local stroage */
        this.load_LS();

        /* 닫기 버튼 클릭시 */
        this.$$close.forEach(this.add_on_click_close);

        /* 스크롤시 */
        window.addEventListener("scroll", this.on_scroll);
    }//init_ls

    /**
     * local stroage에서 불러와가지고 띠배너 보일지 말지
     */
    load_LS() {
        const value = window.localStorage.getItem(this.keyLS);
        // console.log(value);
        const $wrapFix = document.querySelector(".stripbnr-wrap.fixed");
        const $wrapRel = document.querySelector(".stripbnr-wrap.rel");
        if (!$wrapFix) { return; }
        if (!value) {
            window.localStorage.setItem(this.keyLS, "");
            $wrapFix.classList.remove("hide");
            return;
        } else {
            $wrapRel.classList.remove("hide");
            value.split(",").forEach(key => {
                const $bnr = $wrapFix.querySelector(`.stripbnr[data-id="${key}"]`);
                if ($bnr) { $wrapFix.removeChild($bnr); }
            });
            if ($wrapFix.childElementCount === 0) {
                $wrapFix.remove();
            }
        }
    }//load_LS

    /**
     * 
     */
    set_LS(key) {
        const value = window.localStorage.getItem(this.keyLS);
        if (value === null) {
            window.localStorage.setItem(this.keyLS, "");
        }
        const values = value.split(",").filter(str => str);
        values.push(key);
        window.localStorage.setItem(this.keyLS, values.join(","));
    }//load_LS

    /* ============================================= */
    /* ============================================= */
    /* ============================================= */

    /**
     * 쿠키 가져와서...
     */
    load_cookies = () => {
        const $wrapFix = document.querySelector(".stripbnr-wrap.fixed");
        const $wrapRel = document.querySelector(".stripbnr-wrap.rel");

        const $bnr = $wrapFix.querySelector(".stripbnr");
        const key = $bnr.dataset.id;
        const toHide = this.get_cookie(key);

        if (!toHide) {
            // $wrapFix.classList.remove("hide");
            return;
        }

        $wrapRel.classList.remove("hide");
        $wrapFix.remove();
    }//load_cookies

    /**
    * 쿠키 설정 관련 : 쿠키 설정
    * @param {*} param0 
    */
    set_cookie = ({ key, value, expiredays }) => {
        const today = new Date();
        today.setDate(today.getDate() + expiredays);
        document.cookie = key + "=" + encodeURIComponent(value) + "; path=/; expires=" + today.toGMTString() + ";"
    }//set_cookie

    /**
     * 쿠키 설정 관련 : 쿠키 가져오기
     */
    get_cookie = (key) => {
        const cookieData = document.cookie;
        // console.log(cookieData);
        if (cookieData.indexOf(`${key}=done`) < 0) {
            console.log(`get_cookie:${key} 없음`);
            return false;
        }
        const $bnr = document.querySelector(`.fixed .stripbnr[data-id="${key}"]`);
        if (!$bnr) {
            throw new Error("can't find strip banner");
        }
        return true;
    }//get_cookie

    /**
     * 쿠키 설정 관련 : 쿠키 삭제
     * @param {*} key 
     */
    delete_cookie = (key) => {
        document.cookie = `${key}=; max-age=0; path=/;`;
        console.log(key, document.cookie);
    }//delete_cookie

    /* ============================================= */
    /* ============================================= */
    /* ============================================= */

    /**
     * 
     */
    set_dom = () => {
        this.$$close = document.querySelectorAll(".stripbnr-close");
        this.$ftr = document.getElementById("ftr");
    }//set_dom

    /**
     * 닫기 버튼 클릭시
     * @param {*} e 
     */
    add_on_click_close = ($close) => {
        $close.addEventListener("click", () => {
            const key = $close.dataset.targetId;
            console.log(key);
            /* 로컬 스토리지 방식 */
            // this.set_LS(key);

            /* 쿠키 방식 */
            this.set_cookie({
                key,
                value: "done",
                expiredays: 7
            });

            const $wrapFix = $close.parentElement.parentElement.parentElement;
            $wrapFix.remove();

            const $wrapRel = document.querySelector(".stripbnr-wrap.rel");
            $wrapRel.classList.remove("hide");
        }, { once: true });
    }//add_on_click_close

    /**
     * 
    */
    on_scroll = () => {
        if (!this.$ftr) { this.set_dom(); }
        const $wrapRel = document.querySelector(".stripbnr-wrap.rel");
        const $wrapFix = document.querySelector(".stripbnr-wrap.fixed");
        if (window.innerHeight < this.$ftr.getBoundingClientRect().top) {
            /* 푸터가 안 보일때 */
            if ($wrapFix) {
                $wrapRel.classList.add("hide");
                $wrapFix.classList.remove("hide");
            }
            return;
        } else {
            /* 푸터가 보이면 rel은 */
            $wrapRel.classList.remove("hide");
            if ($wrapFix) {
                $wrapFix.classList.add("hide");
            }
        }
    }//on_scroll
}//class:StripBnrCtrl
