    ``` js
    const enable_paste_input_date = () => {
    const $$date = document.querySelectorAll('input[type="date"]');
    if (!$$date.length) {
        requestAnimationFrame(enable_paste_input_date);
        return;
    }
    $$date.forEach($date => {
        $date.addEventListener('paste', (e) => {
            e.preventDefault();

            // 클립보드에서 텍스트 가져오기
            const clipboardData = e.clipboardData.getData('text').trim();

            // 숫자와 구분자 추출 (예: 2026-08-04, 2026.08.04, 20260804 대응)
            const cleaned = clipboardData.replace(/[^\d]/g, '');

            if (cleaned.length === 8) {
                const formattedDate = `${cleaned.slice(0, 4)}-${cleaned.slice(4, 6)}-${cleaned.slice(6, 8)}`;
                e.target.value = formattedDate;
            }
        });
    });
    }
    enable_paste_input_date();
    ```
