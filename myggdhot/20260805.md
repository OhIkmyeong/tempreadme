```js
/* import에 추가(9행) */

/* 공통:GNB(19행 추가) */

```
2. `js/controller/components/` 경로에 `GnbCtrl.js` **파일 생성하여 추가**
```js
/**
 * GNB Controller
 */
export class GnbCtrl{
    /**
     * 
     */
    init = () =>{
        const $$li = document.querySelectorAll(".gnbitem");
        if(!$$li || !$$li.length){
            requestAnimationFrame(()=>{
                this.init();
                return;
            });
        }

        const path = window.location.pathname;

        if(path.includes("about") || path.includes("myggdHotAbout")){
            $$li?.[0].classList.add("on");
        }else if(path.includes("archive") || path.includes("myggdHotList") || path.includes("myggdHotDetail")){
            $$li?.[1].classList.add("on");
        }
    }//init
}//class:GnbCtrl
```
